## Overview

`BarChartOne` renders a vertical bar chart using [ApexCharts](https://apexcharts.com/) via the `react-apexcharts` wrapper. Like the line chart, it is dynamically imported with `ssr: false` to prevent hydration errors caused by ApexCharts' dependency on browser-only APIs.

The chart plots a single **Sales** series across twelve monthly categories with rounded column tops and a transparent stroke between bars.

<Note>
  `BarChartOne` requires the `react-apexcharts` and `apexcharts` packages. Install them if they are not already present:

  ```bash
  npm install react-apexcharts apexcharts
  ```
</Note>

## Import

```tsx
import BarChartOne from "@/components/charts/bar/BarChartOne";
```

## Usage

Drop `BarChartOne` into any page or card component. It manages its own chart options and series data internally.

```tsx
import BarChartOne from "@/components/charts/bar/BarChartOne";
import ComponentCard from "@/components/common/ComponentCard";

export default function BarChartPage() {
  return (
    <div className="space-y-6">
      <ComponentCard title="Bar Chart 1">
        <BarChartOne />
      </ComponentCard>
    </div>
  );
}
```

## Component source

```tsx
"use client";
import React from "react";
import { ApexOptions } from "apexcharts";
import dynamic from "next/dynamic";

// Dynamically import to avoid SSR issues with browser-only ApexCharts APIs
const ReactApexChart = dynamic(() => import("react-apexcharts"), {
  ssr: false,
});

export default function BarChartOne() {
  const options: ApexOptions = {
    colors: ["#465fff"],
    chart: {
      fontFamily: "Outfit, sans-serif",
      type: "bar",
      height: 180,
      toolbar: {
        show: false,
      },
    },
    plotOptions: {
      bar: {
        horizontal: false,
        columnWidth: "39%",
        borderRadius: 5,
        borderRadiusApplication: "end",
      },
    },
    dataLabels: {
      enabled: false,
    },
    stroke: {
      show: true,
      width: 4,
      colors: ["transparent"],
    },
    xaxis: {
      categories: ["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"],
      axisBorder: { show: false },
      axisTicks: { show: false },
    },
    legend: {
      show: true,
      position: "top",
      horizontalAlign: "left",
      fontFamily: "Outfit",
    },
    yaxis: {
      title: {
        text: undefined,
      },
    },
    grid: {
      yaxis: {
        lines: { show: true },
      },
    },
    fill: {
      opacity: 1,
    },
    tooltip: {
      x: { show: false },
      y: {
        formatter: (val: number) => `${val}`,
      },
    },
  };

  const series = [
    {
      name: "Sales",
      data: [168, 385, 201, 298, 187, 195, 291, 110, 215, 390, 280, 112],
    },
  ];

  return (
    <div className="max-w-full overflow-x-auto custom-scrollbar">
      <div id="chartOne" className="min-w-[1000px]">
        <ReactApexChart
          options={options}
          series={series}
          type="bar"
          height={180}
        />
      </div>
    </div>
  );
}
```

## Chart configuration

### Series data

The `series` array contains one object per data group. Each object needs a `name` and a `data` array matching the length of the x-axis `categories`.

```tsx
const series = [
  {
    name: "Sales",
    data: [168, 385, 201, 298, 187, 195, 291, 110, 215, 390, 280, 112],
  },
];
```

### X-axis categories

Categories label each bar along the x-axis. The default uses month abbreviations for a yearly view.

```tsx
xaxis: {
  categories: ["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"],
}
```

### Bar colors

The `colors` array sets the fill color for each series. The default uses TailAdmin's brand blue.

```tsx
colors: ["#465fff"],
```

### Column shape

`plotOptions.bar` controls the visual geometry of each column. TailAdmin uses a `39%` column width with `5px` rounded top corners.

```tsx
plotOptions: {
  bar: {
    horizontal: false,          // vertical bars
    columnWidth: "39%",         // column width as a percentage of available space
    borderRadius: 5,            // rounded corners in pixels
    borderRadiusApplication: "end", // apply radius only to the top of each bar
  },
},
```

### Transparent stroke gap

A `4px` transparent stroke between columns creates a subtle visual gap without affecting bar width.

```tsx
stroke: {
  show: true,
  width: 4,
  colors: ["transparent"],
},
```

### Tooltip formatter

The y-axis tooltip uses a custom formatter to control how values are displayed. Extend this to add units or formatting:

```tsx
tooltip: {
  x: { show: false },
  y: {
    formatter: (val: number) => `$${val.toLocaleString()}`,
  },
},
```

## Real-world example: Monthly Sales

The `MonthlySalesChart` component on the ecommerce dashboard is a direct use of the same configuration as `BarChartOne`, embedded inside a card with a dropdown menu for additional actions.

```tsx
"use client";
import { ApexOptions } from "apexcharts";
import dynamic from "next/dynamic";
import { MoreDotIcon } from "@/icons";
import { DropdownItem } from "@/components/ui/dropdown/DropdownItem";
import { useState } from "react";
import { Dropdown } from "@/components/ui/dropdown/Dropdown";

const ReactApexChart = dynamic(() => import("react-apexcharts"), {
  ssr: false,
});

export default function MonthlySalesChart() {
  const options: ApexOptions = {
    colors: ["#465fff"],
    chart: {
      fontFamily: "Outfit, sans-serif",
      type: "bar",
      height: 180,
      toolbar: { show: false },
    },
    plotOptions: {
      bar: {
        horizontal: false,
        columnWidth: "39%",
        borderRadius: 5,
        borderRadiusApplication: "end",
      },
    },
    dataLabels: { enabled: false },
    stroke: { show: true, width: 4, colors: ["transparent"] },
    xaxis: {
      categories: ["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"],
      axisBorder: { show: false },
      axisTicks: { show: false },
    },
    legend: { show: true, position: "top", horizontalAlign: "left", fontFamily: "Outfit" },
    yaxis: { title: { text: undefined } },
    grid: { yaxis: { lines: { show: true } } },
    fill: { opacity: 1 },
    tooltip: {
      x: { show: false },
      y: { formatter: (val: number) => `${val}` },
    },
  };

  const series = [
    {
      name: "Sales",
      data: [168, 385, 201, 298, 187, 195, 291, 110, 215, 390, 280, 112],
    },
  ];

  const [isOpen, setIsOpen] = useState(false);

  return (
    <div className="overflow-hidden rounded-2xl border border-gray-200 bg-white px-5 pt-5 dark:border-gray-800 dark:bg-white/[0.03] sm:px-6 sm:pt-6">
      <div className="flex items-center justify-between">
        <h3 className="text-lg font-semibold text-gray-800 dark:text-white/90">
          Monthly Sales
        </h3>
        <div className="relative inline-block">
          <button onClick={() => setIsOpen(!isOpen)} className="dropdown-toggle">
            <MoreDotIcon className="text-gray-400 hover:text-gray-700 dark:hover:text-gray-300" />
          </button>
          <Dropdown isOpen={isOpen} onClose={() => setIsOpen(false)} className="w-40 p-2">
            <DropdownItem onItemClick={() => setIsOpen(false)}
              className="flex w-full font-normal text-left text-gray-500 rounded-lg hover:bg-gray-100 hover:text-gray-700 dark:text-gray-400 dark:hover:bg-white/5 dark:hover:text-gray-300">
              View More
            </DropdownItem>
            <DropdownItem onItemClick={() => setIsOpen(false)}
              className="flex w-full font-normal text-left text-gray-500 rounded-lg hover:bg-gray-100 hover:text-gray-700 dark:text-gray-400 dark:hover:bg-white/5 dark:hover:text-gray-300">
              Delete
            </DropdownItem>
          </Dropdown>
        </div>
      </div>
      <div className="max-w-full overflow-x-auto custom-scrollbar">
        <div className="-ml-5 min-w-[650px] xl:min-w-full pl-2">
          <ReactApexChart options={options} series={series} type="bar" height={180} />
        </div>
      </div>
    </div>
  );
}
```

## SSR and dynamic import

ApexCharts depends on `window` and `document` APIs that are not available during server-side rendering. The component handles this with Next.js `dynamic`:

```tsx
import dynamic from "next/dynamic";

const ReactApexChart = dynamic(() => import("react-apexcharts"), {
  ssr: false,
});
```

<Warning>
  Always use `ssr: false` when importing `react-apexcharts` in a Next.js app. Removing this flag will cause a `ReferenceError: window is not defined` error during server rendering.
</Warning>

The file also includes the `"use client"` directive at the top, which is required because the component uses client-side rendering.
